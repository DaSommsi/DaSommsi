// Calculates required length for 6 linear actuators given vehicle pose (surge, sway, heave, roll, pitch, yaw)
void StewartPlatformSolver::computeActuatorLengths(const Pose6DoF& pose, float actuatorLengths[6]) {
    const Matrix3f R = pose.computeRotationMatrix();
    for (size_t i = 0; i < 6; ++i) {
        Vector3f platformPointWorld = pose.position + R * _basePlatformAnchors[i];
        Vector3f legVector = platformPointWorld - _baseFrameAnchors[i];
        actuatorLengths[i] = legVector.norm();
    }
}
