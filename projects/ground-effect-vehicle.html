STM32H7 DMA Optical Sensor Reader Excerpt

void SensorManager_ReadHeightOptical(OpticalSensor_t* dev) {
    uint8_t buffer[6];
    HAL_I2C_Master_Receive_DMA(&hi2c1, dev->i2c_addr, buffer, 6);
    dev->lastHeight_mm = ((uint16_t)buffer[0] << 8 | buffer[1]) * 0.05f;
    dev->isValid = (dev->lastHeight_mm > 0.5f && dev->lastHeight_mm < 25.0f);
}
