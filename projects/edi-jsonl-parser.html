std::string_view EdiParser::nextSegment(std::string_view& stream, char delimiter) noexcept {
    const size_t pos = stream.find(delimiter);
    if (pos == std::string_view::npos) {
        std::string_view tok = stream;
        stream = {};
        return tok;
    }
    std::string_view tok = stream.substr(0, pos);
    stream.remove_prefix(pos + 1);
    return tok;
}
